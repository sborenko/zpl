
При пересчёте больничных сбивается признак совместительства

1. Найдены записи со сбившимся признаком

select distinct tabn, nr
...
select distinct tabn
from z_nu nn
where
  nn.zamesgod >= '1/1/2025'
  and exists (
    select 'x'
    from z_nu n
      join z_otkl o on o.spec_nr = n.spec_nr 
        and o.zamesgod = n.zamesgod
        and o.summa = -n.summa 
    where 
      n.nr = nn.nr
      and n.zamesgod = nn.zamesgod
      and n.vidn in ('2603', '2604', '2605', '2606')
      and n.summa <> 0
      and o.sovm = 'Y'
)
