select TabN, 
  u.Fam || ' ' || substring(u.Ima from 1 for 1) || '.' || substring(u.Otch from 1 for 1) n, 
  'Erroneously indexed' 
from z_sps s
  left join z_UPI u on u.TabN = s.TabN
where dtproc = '08/01/2024' 
  and exists (
    select 'x' 
    from z_EmplIndxExcl
    where TabN = s.TabN
          and (DtBeg is null or DtBeg = "30-DEC-1899" or DtBeg <= '08/31/2024')
          and (DtEnd is null or DtEnd = "30-DEC-1899" or DtEnd >= '08/01/2024')
  )
  and exists (
    select 'x' 
    from z_Otkl
    where Tabn = s.TabN
      and VidN = '1620'
  )
union
select TabN, 
  u.Fam || ' ' || substring(u.Ima from 1 for 1) || '.' || substring(u.Otch from 1 for 1) n, 
  'Not indexed' 
from z_sps s
  left join z_UPI u on u.TabN = s.TabN
where dtproc = '08/01/2024' 
  and not exists (
    select 'x' 
    from z_EmplIndxExcl
    where TabN = s.TabN
          and (DtBeg is null or DtBeg = "30-DEC-1899" or DtBeg <= '08/31/2024')
          and (DtEnd is null or DtEnd = "30-DEC-1899" or DtEnd >= '08/01/2024')
  )
  and not exists (
    select 'x' 
    from z_Otkl
    where Tabn = s.TabN
      and VidN = '1620'
  )
order by 3, 1
