CREATE OR ALTER VIEW Z_VINADB(
    NR,
    KDSPS,
    SPSRSCH,
    TABN,
    VIDN,
    SHVIDN,
    GRP,
    SUMPROC,
    SP,
    PRCSOVM,
    DT_BEG,
    DT_END,
    KDTEMU,
    KDOBJ,
    SHOBJ,
    KDKATP,
    SHKATP,
    PRCHS,
    PRSMM,
    KDSPZ,
    SCHBU,
    SPEC_NR,
    NOTES,
    NRPRBEG,
    DTPRBEG,
    NRPREND,
    DTPREND,
    NRPKT,
    DTVPKT,
    PRC_WT,
    ZP13,
    OTP,
    BLN,
    DKR,
    OTPUSKN,
    BOLNIC,
    SPRM,
    CPRM,
    SSRK,
    DETI,
    DETIM,
    DETIK,
    DETIG,
    IND,
    DHD,
    OKLFND,
    CBUDG,
    SNDB,
    SPDN,
    SWSB,
    SPRF,
    SPENS,
    SALIM,
    CNDB,
    SSOCSTRAH,
    SBZR,
    SP_PF,
    SP_FSTVP,
    SP_FSNV,
    SP_FZ,
    SOVM,
    SPOV,
    KEKV,
    KDSCHDT,
    KDSCHKT,
    KD_DLZHN,
    KD_SOVM,
    IS_PERE,
    SPOV_MERGE,
    PARTSALID,
    SPOV_BEGDT,
    SPOV_ENDDT
) AS
Select n.NR, n.KDSPS, n.SPSRSCH, n.TABN, n.VIDN, v.SHVIDN, 
       n.GRP, n.SUMPROC, n.SP, n.PRCSOVM, 
       n.DT_BEG, n.DT_END, n.KdTemu, n.KDOBJ, o.ShObj, n.KDKATP, kp.ShKatP, n.PRCHS, n.PRSMM,
       n.KDSPZ, s. Sch_17,  n.SPEC_NR,
       n.NOTES, n.NRPRBEG, n.DTPRBEG, n.NRPREND, n.DTPREND, n.NRPKT, n.DTVPKT, n.PRC_WT,
       v.ZP13, v.OTP, v.BLN, v.DKR, v.OTPUSKN, v.BOLNIC, v.SPRM, v.CPRM,
       v.SSRK, v.DETI, v.DETIM, v.DETIK, v.DETIG, v.IND, v.DHD, v.OKLFND, v.CBUDG,
       v.SNDB, v.SPDN, v.SWSB, v.SPRF, v.SPENS, v.SALIM, v.CNDB, v.SSOCSTRAH, v.SBZR,
       v.SP_PF, v.SP_FSTVP, v.SP_FSNV, v.SP_FZ, v.SOVM, v.SPOV, v.KEKV,
       v.KDSCHDT, v.KDSCHKT, n.KD_DLZHN, n.KD_SOVM, n.IS_PERE, n.SPOV_MERGE, n.PARTSALID,
       n.SPOV_BEGDT, n.SPOV_ENDDT
from z_Nadb n 
   Left Outer Join z_KdObj o on o.KdObj = n.KdObj
   Left Outer Join z_KatPers kp on kp.KdKatP = n.KdKatP
   Left Outer Join Kd_Sbu s on s.KdSch = n.KdSpz
   Left Outer Join z_KdVidN v on v.VidN = n.VidN
;
